export type TaskStatus = 'queued' | 'running' | 'paused' | 'completed' | 'failed' | 'cancelled';
export interface TaskSummary { id: string; agentId?: string; status: TaskStatus; createdAt: string; completedAt?: string; error?: string }
export interface BoardColumns { queued: TaskSummary[]; running: TaskSummary[]; paused: TaskSummary[]; completed: TaskSummary[]; failed: TaskSummary[]; cancelled: TaskSummary[] }

export function boardOrder(): TaskStatus[] { return ['queued', 'running', 'paused', 'completed', 'failed', 'cancelled']; }

export function groupByStatus(tasks: TaskSummary[]): BoardColumns {
  const cols: BoardColumns = { queued: [], running: [], paused: [], completed: [], failed: [], cancelled: [] };
  for (const task of tasks) cols[task.status]?.push(task);
  for (const status of boardOrder()) cols[status].sort((a, b) => a.createdAt.localeCompare(b.createdAt));
  return cols;
}
```

export interface Usage { promptTokens: number; completionTokens: number; totalTokens: number }
export interface UsageRecord {
  taskId?: string; sessionId?: string; agentId?: string; modelId?: string;
  promptTokens: number; completionTokens: number; totalTokens: number;
  costEstimate?: number; createdAt?: string;
}

export function estimateTokens(text: string): number {
  let cjk = 0, ascii = 0;
  for (const ch of text) (ch.codePointAt(0)! > 0x2e7f ? cjk++ : ascii++);
  return Math.max(1, Math.ceil(ascii / 4 + cjk * 1.5));
}
export function estimateUsage(prompt: string, completion: string): Usage {
  return { promptTokens: estimateTokens(prompt), completionTokens: estimateTokens(completion), totalTokens: estimateTokens(prompt) + estimateTokens(completion) };
}
export function sumUsage(records: Pick<UsageRecord, 'promptTokens' | 'completionTokens' | 'totalTokens'>[]): Usage & { calls: number } {
  const s = { promptTokens: 0, completionTokens: 0, totalTokens: 0 };
  for (const r of records) { s.promptTokens += r.promptTokens; s.completionTokens += r.completionTokens; s.totalTokens += r.totalTokens; }
  return { ...s, calls: records.length };
}
export function estimateCost(records: UsageRecord[], price: { inputPerM: number; outputPerM: number }): number {
  return records.reduce((a, r) => a + (r.promptTokens * price.inputPerM + r.completionTokens * price.outputPerM) / 1_000_000, 0);
}
```

import type { Database } from 'better-sqlite3';
import { sumUsage, type UsageRecord, type Usage } from '@jarvis/core';

export interface UsageSummary { total: Usage & { calls: number }; byAgent: Array<{ agentId: string; usage: Usage & { calls: number } }> }

export class UsageTracker {
  private insert;
  private select;
  constructor(db: Database) {
    this.insert = db.prepare(`INSERT INTO token_usage (task_id, session_id, agent_id, model_id, prompt_tokens, completion_tokens, total_tokens, cost_estimate)
      VALUES (@taskId, @sessionId, @agentId, @modelId, @promptTokens, @completionTokens, @totalTokens, @costEstimate)`);
    this.select = db.prepare('SELECT * FROM token_usage ORDER BY created_at');
  }
  track(r: UsageRecord): void { this.insert.run({ ...r, costEstimate: r.costEstimate ?? null }); }
  summary(): UsageSummary {
    const rows = this.select.all() as UsageRecord[];
    const byAgent = new Map<string, Usage & { calls: number }>();

```ts
import { mkdirSync, statSync, readdirSync, copyFileSync } from 'node:fs';
import { join, basename } from 'node:path';
import type { Database } from 'better-sqlite3';

export interface BackupInfo { file: string; name: string; sizeBytes: number; createdAt: string }

export class BackupService {
  private timer: NodeJS.Timeout | null = null;
  constructor(private db: Database, private dir: string, private mainPath?: string) { mkdirSync(dir, { recursive: true }); }

  start(intervalMs = 24 * 60 * 60 * 1000): void {
    this.stop();
    this.timer = setInterval(() => { void this.createBackup(); }, intervalMs);
    this.timer.unref();
  }
  stop(): void { if (this.timer) { clearInterval(this.timer); this.timer = null; } }

  async createBackup(): Promise<string> {
    const stamp = new Date().toISOString().replace(/[-:.T]/g, '').slice(0, 14);
    const file = join(this.dir, `${stamp}.db`);
    await this.db.backup(file); // better-sqlite3 在线备份(含 WAL 一致快照)
    return file;
  }
  list(): BackupInfo[] {
    return readdirSync(this.dir).filter(f => f.endsWith('.db')).map(f => {
      const p = join(this.dir, f); const st = statSync(p);
      return { file: p, name: f, sizeBytes: st.size, createdAt: st.birthtime.toISOString() };
    }).sort((a, b) => b.createdAt.localeCompare(a.createdAt));
  }
  async restore(file: string): Promise<void> {
    if (!this.mainPath) throw new Error('mainPath required for restore');
    this.db.pragma('wal_checkpoint(TRUNCATE)');
    this.db.close();
    copyFileSync(file, this.mainPath);
  }
}
```
> `db.backup(dest)` 为 better-sqlite3 内建异步在线备份;spec 中无需 `mainPath`。

```ts
import { rmSync } from 'node:fs';
import type { Database } from 'better-sqlite3';
import { confirmPhrase, DEFAULT_WIPE_TABLES, type WipeScope, type WipeResult } from '@jarvis/core';

export class WipeService {
  constructor(private db: Database, private keychain: { deleteAllApiKeys: () => number }, private workspaceRoot?: string) {}
  wipe(scope: WipeScope, phrase: string): WipeResult {
    if (phrase !== confirmPhrase(scope)) throw new Error('confirmation phrase mismatch');
    const deleted: Record<string, number> = {};
    for (const table of scope.tables) {
      if (!DEFAULT_WIPE_TABLES.includes(table)) continue;
      const info = this.db.prepare(`DELETE FROM ${table}`).run();
      deleted[table] = info.changes;
    }
    const keychainDeleted = scope.keychain ? this.keychain.deleteAllApiKeys() : 0;
    let workspaceRemoved = false;
    if (scope.workspace && this.workspaceRoot) { rmSync(this.workspaceRoot, { recursive: true, force: true }); workspaceRemoved = true; }
    this.db.pragma('wal_checkpoint(TRUNCATE)');
    this.db.exec('VACUUM');
    return { deleted, keychainDeleted, workspaceRemoved, vacuumed: true };
  }
}

export const CURRENT_SCHEMA = 8; // = 当前迁移版本(M8 → v8)
export interface ProviderExport { id: string; type: string; baseUrl: string; apiKeyRef?: string; [k: string]: unknown }
export interface ModelExport { id: string; providerId: string; modelId: string; name: string }
export interface AgentExport { id: string; name: string; slug: string; description?: string; systemPrompt?: string; modelId?: string; [k: string]: unknown }
export interface ExportPayload { schemaVersion: number; exportedAt: string; providers: ProviderExport[]; models: ModelExport[]; agents: AgentExport[]; settings: Record<string, unknown> }
export type ImportStrategy = 'skip' | 'overwrite' | 'merge';
export interface ImportPlan { create: ExportPayload['providers' | 'models' | 'agents']; update: ExportPayload['providers' | 'models' | 'agents']; skip: string[] }

export function buildExport(providers: Array<{ id: string; type: string; base_url: string; api_key_ref?: string | null }>, models: ModelExport[], agents: AgentExport[], settings: Record<string, unknown>): ExportPayload {
  return {
    schemaVersion: CURRENT_SCHEMA,
    exportedAt: new Date().toISOString(),
    providers: providers.map(p => ({ id: p.id, type: p.type, baseUrl: p.base_url, apiKeyRef: p.api_key_ref ?? undefined })),
    models, agents, settings,
  };
}
export function validateSchema(p: ExportPayload): { ok: true } | { ok: false; error: string } {
  if (typeof p.schemaVersion !== 'number' || p.schemaVersion < 1) return { ok: false, error: 'missing schemaVersion' };
  if (p.schemaVersion > CURRENT_SCHEMA) return { ok: false, error: `schema ${p.schemaVersion} > current ${CURRENT_SCHEMA}` };
  return { ok: true };
}
export function mergeEntity<T extends { id: string }>(existing: T, incoming: Partial<T>): T {
  const out = { ...existing };
  for (const [k, v] of Object.entries(incoming)) if (v !== undefined && v !== null && v !== '') (out as Record<string, unknown>)[k] = v;
  return out;
}
export function planImport(p: ExportPayload, current: { providers: ProviderExport[]; agents: AgentExport[] }, strategy: ImportStrategy): ImportPlan {
  const byId = (coll: Array<{ id: string }>) => new Map(coll.map(x => [x.id, x]));
  const provCur = byId(current.providers); const agentCur = byId(current.agents);
  const plan: ImportPlan = { create: [], update: [], skip: [] };
  const decide = (id: string, incoming: ProviderExport | AgentExport, cur: Map<string, any>) => {
    if (!cur.has(id)) { plan.create.push(incoming); return; }
    if (strategy === 'skip') { plan.skip.push(id); return; }
    plan.update.push(strategy === 'merge' ? mergeEntity(cur.get(id), incoming) : incoming);
  };
  for (const p of p.providers) decide(p.id, p, provCur);
  for (const a of p.agents) decide(a.id, a, agentCur);
  return plan;
}

import { parse as yamlParse, stringify as yamlStringify } from 'yaml';
import type { Database } from 'better-sqlite3';
import { buildExport, planImport, validateSchema, type ExportPayload, type ImportStrategy } from '@jarvis/core';

export function createConfigIpc(db: Database, settingsGet: (k: string) => unknown) {
  const loadPayload = (): ExportPayload => {
    const providers = db.prepare('SELECT id, type, base_url, api_key_ref FROM providers').all() as Array<Record<string, unknown>>;
    const models = db.prepare('SELECT id, provider_id AS providerId, model_id AS modelId, name FROM models').all() as ExportPayload['models'];
    const agents = db.prepare('SELECT id, name, slug, description, system_prompt AS systemPrompt, model_id AS modelId FROM agents').all() as ExportPayload['agents'];
    return buildExport(providers as never, models, agents, (settingsGet('__all__') ?? {}) as Record<string, unknown>);
  };
  const exportConfig = (format: 'json' | 'yaml' = 'json') => format === 'yaml' ? yamlStringify(loadPayload()) : JSON.stringify(loadPayload(), null, 2);
  const importConfig = (text: string, strategy: ImportStrategy) => {
    const payload = (text.trim().startsWith('{') ? JSON.parse(text) : yamlParse(text)) as ExportPayload;
    const chk = validateSchema(payload); if (!chk.ok) return { ok: false as const, error: chk.error };
    const current = { providers: db.prepare('SELECT id, type, base_url AS baseUrl FROM providers').all() as never[], agents: db.prepare('SELECT id, name, slug FROM agents').all() as never[] };
    const plan = planImport(payload, current, strategy);
    const insP = db.prepare('INSERT INTO providers (id, type, base_url, api_key_ref) VALUES (?,?,?,?)');
    const insA = db.prepare('INSERT INTO agents (id, name, slug) VALUES (?,?,?)');
    for (const p of plan.create as ExportPayload['providers']) insP.run(p.id, p.type, p.baseUrl, p.apiKeyRef ?? null);
    for (const a of plan.create as ExportPayload['agents']) insA.run(a.id, a.name, a.slug);
    return { ok: true as const, created: plan.create.length, updated: plan.update.length, skipped: plan.skip.length };
  };
  return { exportConfig, importConfig };
}

export function normalizeCombo(e: { key: string; ctrlKey?: boolean; metaKey?: boolean; shiftKey?: boolean; altKey?: boolean }): string {
  const mods: string[] = [];
  if (e.metaKey) mods.push('Cmd');
  if (e.ctrlKey) mods.push('Ctrl');

}

import { lookup, type ShortcutAction } from '@jarvis/core';

export function useShortcuts(handlers: Partial<Record<ShortcutAction, () => void>>) {
  const ref = useRef(handlers);
  ref.current = handlers;
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      // 输入框内保留 Esc 取消 / Enter 发送由组件自行处理,避免双重触发
      if ((e.target as HTMLElement)?.tagName === 'INPUT' && !e.metaKey && !e.ctrlKey && !e.altKey && e.key === 'Escape') return;
      void (async () => {
        const bindings = (await window.jarvis.invoke('shortcuts.get')) as Record<ShortcutAction, string>;
        const action = lookup(bindings, normalizeComboLite(e));
        if (action && ref.current[action]) { e.preventDefault(); ref.current[action]!(); }
      })();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);
}

import { create } from 'zustand';
import type { Workflow } from '@jarvis/core';

export interface WfUiNode { id: string; agentId: string; input: string }
export interface WfUiEdge { id: string; from: string; to: string }

export function toWorkflow(nodes: WfUiNode[], edges: WfUiEdge[]): Workflow {
  const ids = new Set(nodes.map(n => n.id));
  for (const e of edges) if (!ids.has(e.from) || !ids.has(e.to)) throw new Error('missing node ' + (ids.has(e.from) ? e.to : e.from));
  return { nodes, edges };
}

let seq = 0;
const nextId = (p: string) => `${p}-${Date.now()}-${seq++}`;

export const useWorkflowStore = create<{
  nodes: WfUiNode[]; edges: WfUiEdge[]; agents: Array<{ id: string; name: string }>;
  outputs: Record<string, string> | null;
  loadAgents: () => Promise<void>;
  addNode: (agentId: string) => void;
  removeNode: (id: string) => void;
  connect: (from: string, to: string) => void;
  setInput: (id: string, input: string) => void;
  run: () => Promise<void>;
}>((set, get) => ({
  nodes: [], edges: [], agents: [], outputs: null,
  loadAgents: async () => set({ agents: (await window.jarvis.invoke('agent.list')) as Array<{ id: string; name: string }> }),
  addNode: (agentId) => set(s => ({ nodes: [...s.nodes, { id: nextId('n'), agentId, input: '' }] })),
  removeNode: (id) => set(s => ({ nodes: s.nodes.filter(n => n.id !== id), edges: s.edges.filter(e => e.from !== id && e.to !== id) })),
  connect: (from, to) => set(s => ({ edges: [...s.edges, { id: nextId('e'), from, to }] })),
  setInput: (id, input) => set(s => ({ nodes: s.nodes.map(n => n.id === id ? { ...n, input } : n) })),
  run: async () => {
    const wf = toWorkflow(get().nodes, get().edges);
    const outputs = (await window.jarvis.invoke('workflow.run', { definitionJson: JSON.stringify(wf) })) as Record<string, string>;
    set({ outputs });
  },
}));
```

export type ArtifactKind = 'table' | 'chart' | 'mermaid' | 'markdown';
export interface Artifact { id: string; taskId: string; kind: ArtifactKind; title?: string; content: string }
export interface TableData { headers: string[]; rows: string[][] }
export interface DataPoint { label: string; value: number }

export function parseTable(markdown: string): TableData {
  const lines = markdown.trim().split('\n').filter(l => l.trim().startsWith('|'));
  const cells = (l: string) => l.trim().replace(/^\||\|$/g, '').split('|').map(c => c.trim());
  if (lines.length === 0) return { headers: [], rows: [] };
  const headers = cells(lines[0]);
  const rows = lines.slice(2).map(cells); // 跳过分隔行
  return { headers, rows };
}
export function extractMermaid(text: string): string | null {
  const m = text.match(/```mermaid\s*\n([\s\S]*?)```/);
  return m ? m[1].trim() : null;
}
export function captureArtifacts(taskId: string, resultText: string): Artifact[] {
  const out: Artifact[] = [];
  const tableBlock = resultText.match(/^\|[\s\S]*?\n\|[\s\S]*?(?=\n\n|$)/m);
  if (tableBlock && tableBlock[0].split('\n').length >= 3) out.push({ id: `${taskId}-t1`, taskId, kind: 'table', content: tableBlock[0].trim() });
  const md = extractMermaid(resultText);
  if (md) out.push({ id: `${taskId}-m1`, taskId, kind: 'mermaid', content: md });
  if (out.length === 0 && resultText.trim()) out.push({ id: `${taskId}-md1`, taskId, kind: 'markdown', content: resultText.trim() });
  return out;
}
export function dataPointSeries(records: Array<{ label: string; value: number }>): DataPoint[] { return records; }
